; A/UX Toolbox (user-mode Mac environment) change for the LC III and LC III+, with
; modules/sonora/toolbox.txt (BoxFlag 0x0C for them, and Simple Beep). Apply with
; tools/binpatch.py. Written against A/UX 3.1 (AWS 95) /mac/lib/Patches.

; GetLowFromKernel's dispatch on BoxFlag gives the LC III, 0x15 (Gestalt 27, also the
; Performa 450), and the LC III+, 0x38 (Gestalt 62, also the Performa 460, 466 and 467),
; d1 = 0x0C and the exit that stores d1 in BoxFlag, and the Quadras d1 = 0x10.
;
; The site runs from the redundant second load of BoxFlag into d0 to the jump table: the
; old range check, its bhi, the table dispatch and the 4-byte swbeg header (illegal; dc.w
; 30) that the jmp skips. d1 is a scratch register to C callers, and the only caller,
; doVariables, goes straight on to call InitUniversalPtr and sets d1 itself before using
; it. The range check is the old one with word operations (the table indexes with d0.w),
; and the dispatch is the old one, moved up 4 bytes against the same table.
file /mac/lib/Patches/Patch.067C
at GetLowFromKernel+0x3a   ; was: move.b BoxFlag,d0; extb.l d0; subi.l #$18,d0; cmpi.l #$1d,d0;
                           ;      bhi.b; add.w; move.w (table,pc,d0.w),d0; jmp (table,pc,d0.w);
                           ;      swbeg
check d7b574f7d84fd99b
new 72 0c                  ; moveq   #$c,d1           d0 = BoxFlag already
new 0c 40 00 15            ; cmpi.w  #$15,d0
new 67 c8                  ; beq.b   GetLowFromKernel+0xa (BoxFlag = d1): LC III
new 0c 40 00 38            ; cmpi.w  #$38,d0
new 67 c2                  ; beq.b   GetLowFromKernel+0xa: LC III+
new 72 10                  ; moveq   #$10,d1          for the Quadras in the table
new 04 40 00 18            ; subi.w  #$18,d0
new 0c 40 00 1d            ; cmpi.w  #$1d,d0
new 62 be                  ; bhi.b   GetLowFromKernel+0x12 (leave BoxFlag)
new d0 40                  ; add.w   d0,d0
new 30 3b 00 06            ; move.w  (GetLowFromKernel+0x5e,pc,d0.w),d0
new 4e fb 00 02            ; jmp     (GetLowFromKernel+0x5e,pc,d0.w)
